// Rejected experiment, not shipping code. Requires proxy rule inheritance too.
#[allow(clippy::too_many_arguments)]
// Rive traverses its paint list backwards. Keep import order for layout, and
// explicitly place each earlier sibling subtree after the next in that list.
fn write_sibling_paint_order(children: &[u32], root_start: u32, records: &mut Vec<Record>) -> Result<(), Diagnostic> {
    for pair in children.windows(2) {
        let target_id = records.len() as u32 - root_start;
        let mut target = Record::new("DrawTarget");
        target.set("parentId", Value::Uint(target_id + 1))?;
        target.set("drawableId", Value::Uint(pair[1]))?;
        target.set("placementValue", Value::Uint(1))?;
        records.push(target);
        let mut rules = Record::new("DrawRules");
        rules.set("parentId", Value::Uint(pair[0]))?;
        rules.set("drawTargetId", Value::Uint(target_id))?;
        records.push(rules);
    }
    Ok(())
}
